module uart_tx #(
    parameter CLK_FREQ  = 50_000_000,
    parameter BAUDRATE  = 115_200
)(
    input  logic       clk,
    input  logic       rst_n,
    input  logic       tx_start,
    input  logic [7:0] tx_data,
    output logic       tx,
    output logic       tx_busy,
    output logic       tx_done
);

    localparam CLKS_PER_BIT = CLK_FREQ / BAUDRATE;

    typedef enum logic [1:0] {
        IDLE  = 2'b00,
        START = 2'b01,
        DATA  = 2'b10,
        STOP  = 2'b11
    } state_t;

    state_t state;
    logic [15:0] clk_count;
    logic [2:0]  bit_index;
    logic [7:0]  shifter;

    always_ff @(posedge clk or negedge rst_n) begin
        if (!rst_n) begin
            state     <= IDLE;
            clk_count <= 0;
            bit_index <= 0;
            shifter   <= 8'h00;
            tx        <= 1'b1;
            tx_busy   <= 1'b0;
            tx_done   <= 1'b0;
        end else begin
            tx_done <= 1'b0;
            case (state)
                IDLE: begin
                    tx <= 1'b1;
                    clk_count <= 0;
                    bit_index <= 0;
                    if (tx_start) begin
                        shifter   <= tx_data;
                        tx_busy   <= 1'b1;
                        state     <= START;
                    end else begin
                        tx_busy   <= 1'b0;
                    end
                end
                START: begin
                    tx <= 1'b0;
                    if (clk_count < CLKS_PER_BIT - 1)
                        clk_count <= clk_count + 1;
                    else begin
                        clk_count <= 0;
                        state     <= DATA;
                    end
                end
                DATA: begin
                    tx <= shifter[bit_index];
                    if (clk_count < CLKS_PER_BIT - 1)
                        clk_count <= clk_count + 1;
                    else begin
                        clk_count <= 0;
                        if (bit_index < 7)
                            bit_index <= bit_index + 1;
                        else begin
                            bit_index <= 0;
                            state     <= STOP;
                        end
                    end
                end
                STOP: begin
                    tx <= 1'b1;
                    if (clk_count < CLKS_PER_BIT - 1)
                        clk_count <= clk_count + 1;
                    else begin
                        clk_count <= 0;
                        tx_done   <= 1'b1;
                        tx_busy   <= 1'b0;
                        state     <= IDLE;
                    end
                end
                default: state <= IDLE;
            endcase
        end
    end
endmodule

module tb_uart_tx;
    localparam CLK_FREQ = 10_000_000; 
    localparam BAUDRATE = 1_000_000;  
    localparam BIT_PERIOD_NS = (1_000_000_000 / BAUDRATE);

    logic       clk;
    logic       rst_n;
    logic       tx_start;
    logic [7:0] tx_data;
    logic       tx;
    logic       tx_busy;
    logic       tx_done;

    uart_tx #(
        .CLK_FREQ(CLK_FREQ),
        .BAUDRATE(BAUDRATE)
    ) dut (
        .clk(clk),
        .rst_n(rst_n),
        .tx_start(tx_start),
        .tx_data(tx_data),
        .tx(tx),
        .tx_busy(tx_busy),
        .tx_done(tx_done)
    );

    always #10 clk = ~clk;

    task send_and_verify(input logic [7:0] data_to_send);
        logic [7:0] rx_payload;
        $display("[TB] Sending Data Byte: 0x%0h", data_to_send);
        @(posedge clk);
        tx_data  <= data_to_send;
        tx_start <= 1'b1;
        @(posedge clk);
        tx_start <= 1'b0;

        wait (tx == 1'b0);
        #(BIT_PERIOD_NS / 2);
        if (tx !== 1'b0) $error("[TB ERROR] Invalid Start bit!");

        for (int i = 0; i < 8; i++) begin
            #BIT_PERIOD_NS;
            rx_payload[i] = tx;
        end

        #BIT_PERIOD_NS;
        if (tx !== 1'b1) $error("[TB ERROR] Missing Stop Bit!");

        if (rx_payload === data_to_send)
            $display("[SUCCESS] Transmitted: 0x%0h | Received: 0x%0h\n", data_to_send, rx_payload);
        else
            $error("[FAIL] Transmitted: 0x%0h | Received: 0x%0h\n", data_to_send, rx_payload);
    endtask

    initial begin
        $dumpfile("dump.vcd");
        $dumpvars(0, tb_uart_tx);
        clk      = 0;
        rst_n    = 0;
        tx_start = 0;
        tx_data  = 8'h00;
        #40 rst_n = 1;
        #40;
        send_and_verify(8'hA5);
        send_and_verify(8'h3C);
        send_and_verify(8'hFF);
        $display("===== ALL TESTS COMPLETED SUCCESSFULLY =====");
        $finish;
    end
endmodule
